// Terrain round 3 review shots: S2 as is vs S2 organic (?terrain=s2&organic=1) on a generated map with a
// 128-tile forest, a 19-tile range, a 2-tile range and a 2-tile grove (seed 2914823), at normal and far
// zoom, close-ups, the Settle and Resources lenses, 1000x600, S1/S3 organic, and a frame-time sweep on a
// Large (60x40) map. Shots only. TERRAIN_SHOTS=<dir> picks the output directory.
import { test as base, expect, type Page } from '@playwright/test';
import { createServer } from 'vite';
import { mkdir, writeFile } from 'node:fs/promises';
import path from 'node:path';
import type { useGame } from '../src/store';
import { coreFixture } from '../src/test/coreFixture';
import { mapArtFixture } from '../src/test/mapArtFixture';
import { spawnUnit } from '../src/test/fixtureEngine';
import { hexEq } from '../src/hex';
import type { Hex } from '../src/types';

type Perf = { frames: number; drawMs: number; maxMs: number; workMs: number; buildMs: number; reset(): void };
type MapWindow = Window & { legatum: typeof useGame; legatumMap: { camera: { x: number; y: number; zoom: number }; redraw(): void; perf: Perf } };
const out = process.env.TERRAIN_SHOTS ? path.resolve(process.env.TERRAIN_SHOTS) : path.resolve('test-results/terrain-round3');

const test = base.extend<{}, { devURL: string }>({
  devURL: [async ({}, use) => {
    const server = await createServer({ cacheDir: 'test-results/terrain-round3/vite-cache', server: { host: '127.0.0.1', port: 0 } });
    await server.listen();
    try {
      const address = server.httpServer!.address();
      if (!address || typeof address === 'string') throw new Error('Vite did not bind');
      await use(`http://127.0.0.1:${address.port}`);
    } finally { await server.close(); }
  }, { scope: 'worker' }],
  baseURL: async ({ devURL }, use) => { await use(devURL); },
});

const SEED = '2914823';
const hexOf = (col: number, row: number): Hex => ({ q: col - Math.floor((row - (row & 1)) / 2), r: row });

function scene(size?: { width: number; height: number }) {
  const g = mapArtFixture(SEED, size).fixture;
  const W = g.width;
  const freeNear = (col: number, row: number, ok: string[] = ['Grassland', 'Plains', 'Desert']): Hex => {
    for (let d = 0; d < 8; d++) for (let dr = -d; dr <= d; dr++) for (let dc = -d; dc <= d; dc++) {
      const c = col + dc, r = row + dr;
      if (c < 0 || r < 0 || c >= W || r >= g.height) continue;
      const h = hexOf(c, r);
      if (!ok.includes(g.terrain[r * W + c]) || g.resource?.[r * W + c] || g.wonder?.[r * W + c]) continue;
      if (g.units.some((u) => hexEq(u.pos, h)) || g.cities.some((x) => hexEq(x.pos, h))) continue;
      return h;
    }
    throw new Error('no free hex');
  };
  const settler = spawnUnit(g, 0, 'Settler', freeNear(33, 13, ['Grassland', 'Plains']), false);
  spawnUnit(g, 0, 'Soldier', freeNear(23, 9), false);
  spawnUnit(g, 0, 'Scout', freeNear(34, 9), false);
  spawnUnit(g, 3, 'Soldier', freeNear(35, 15, ['Grassland', 'Plains']), false);
  spawnUnit(g, 5, 'Ranger', freeNear(21, 10, ['Grassland', 'Plains']), false);
  return { g, settler };
}

async function open(page: Page, game: unknown, query: string) {
  await page.addInitScript(({ game }) => {
    localStorage.setItem('legatum.autosave.v1', JSON.stringify({ format: 'legatum-save', version: 1, client: 'wasm', turn: 12, names: ['Ada'], savedAt: new Date().toISOString(), game }));
    localStorage.setItem('legatum.first-game-hints.v1', JSON.stringify({ enabled: false, seen: [] }));
  }, { game });
  await page.goto(`/?client=wasm&${query}`);
  await page.getByRole('button', { name: /^Continue/ }).click();
  await expect(page.getByTestId('turn-number')).toHaveText('12');
  await expect(page.locator('.loading-screen')).toHaveCount(0);
  await page.evaluate(() => document.fonts.ready.then(() => undefined));
}

async function frame(page: Page, at: Hex, zoom: number, select?: number) {
  await page.evaluate(({ at, zoom, select }) => {
    const w = window as unknown as MapWindow;
    const s = w.legatum.getState();
    s.set({ techOpen: false });
    s.select(select ? { kind: 'unit', id: select } : null);
    s.jumpTo(at);
    w.legatumMap.camera.zoom = zoom;
    w.legatumMap.redraw();
  }, { at, zoom, select });
  // Park the mouse on the lens panel's heading: no tooltip there (the top bar's yields and map tiles both show one).
  await page.mouse.move(60, (page.viewportSize()?.height ?? 900) - 107);
  await page.waitForTimeout(1200);
}
async function lens(page: Page, name: string) {
  await page.getByRole('radiogroup', { name: 'Map lens', exact: true }).getByRole('radio', { name: new RegExp(`^${name}`) }).click();
  await page.mouse.move(60, (page.viewportSize()?.height ?? 900) - 107);
  await page.waitForTimeout(800);
}

const VARIANTS: [string, string][] = [['s2', 'terrain=s2'], ['s2o', 'terrain=s2&organic=1']];
const { g, settler } = scene();
const game = coreFixture(g);

for (const [name, query] of VARIANTS) {
  test(`terrain ${name}`, async ({ page }) => {
    test.setTimeout(180_000);
    await page.setViewportSize({ width: 1440, height: 900 });
    await open(page, game, query);
    const shot = async (n: string) => { await mkdir(out, { recursive: true }); await page.screenshot({ path: `${out}/${name}-${n}.png`, animations: 'disabled' }); };
    const centre = hexOf(29, 12);
    await frame(page, centre, 0.66); await shot('normal');
    await frame(page, centre, 0.4); await shot('far');
    await frame(page, hexOf(27, 9), 1.25); await shot('forest');
    await frame(page, hexOf(25, 25), 1.25); await shot('grove');
    await frame(page, hexOf(30, 14), 1.1); await shot('range');
    await frame(page, hexOf(38, 13), 1.25); await shot('smallrange');
    await frame(page, centre, 0.66, settler.id); await lens(page, 'Settle'); await shot('settle');
    await frame(page, centre, 0.66); await lens(page, 'Resources'); await shot('resources');
    await lens(page, 'Default');
    await page.setViewportSize({ width: 1000, height: 600 });
    await frame(page, centre, 0.66); await shot('small');
  });
}

for (const v of ['s1', 's3']) {
  test(`terrain ${v} organic`, async ({ page }) => {
    test.setTimeout(120_000);
    await page.setViewportSize({ width: 1440, height: 900 });
    await open(page, game, `terrain=${v}&organic=1`);
    await frame(page, hexOf(29, 12), 0.66);
    await mkdir(out, { recursive: true });
    await page.screenshot({ path: `${out}/${v}o-normal.png`, animations: 'disabled' });
  });
}

// Frame time on a Large map: a pan sweep across the whole map at normal zoom, then at far zoom, from a cold cache.
const large = coreFixture(scene({ width: 60, height: 40 }).g);
for (const [name, query] of [...VARIANTS, ['current', 'terrain=current']] as [string, string][]) {
  test(`perf ${name}`, async ({ page }) => {
    test.setTimeout(180_000);
    await page.setViewportSize({ width: 1440, height: 900 });
    await open(page, large, query);
    const res = await page.evaluate(async () => {
      const w = window as unknown as MapWindow;
      const m = w.legatumMap;
      const raf = () => new Promise((r) => requestAnimationFrame(() => r(null)));
      const sweep = async (zoom: number) => {
        m.camera.zoom = zoom;
        const xs = [0, 4000];
        const ys = [400, 1000, 1600];
        m.perf.reset();
        const times: number[] = [];
        let last = performance.now();
        for (const y of ys) for (let k = 0; k <= 60; k++) {
          m.camera.x = xs[0] + ((xs[1] - xs[0]) * k) / 60;
          m.camera.y = y;
          m.redraw();
          await raf();
          const now = performance.now();
          times.push(now - last);
          last = now;
        }
        times.sort((a, b) => a - b);
        return { frames: m.perf.frames, drawMsAvg: m.perf.drawMs / Math.max(1, m.perf.frames), drawMsMax: m.perf.maxMs, workMs: m.perf.workMs, frameP50: times[Math.floor(times.length / 2)], frameP95: times[Math.floor(times.length * 0.95)] };
      };
      const buildMs = m.perf.buildMs;
      return { buildMs, normal: await sweep(0.66), far: await sweep(0.4), normalWarm: await sweep(0.66) };
    });
    await mkdir(out, { recursive: true });
    await writeFile(`${out}/perf-${name}.json`, JSON.stringify(res, null, 2));
  });
}
